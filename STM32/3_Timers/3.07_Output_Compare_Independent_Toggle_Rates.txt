3.7  OUTPUT COMPARE INDEPENDENT TOGGLE RATES
============================================

Board      : STM32F407G-DISC1 (Discovery)
Tool       : STM32CubeIDE, HAL
Clock      : HSI 16 MHz (timer clock 16 MHz) unless stated
Debug      : SYS -> Serial Wire, always
Onboard    : PD12 green, PD13 orange, PD14 red, PD15 blue, PA0 user button
             (HIGH when pressed), LIS3DSH accelerometer on SPI1
             (PA5 SCK, PA6 MISO, PA7 MOSI, PE3 CS), internal ADC temp/VREFINT

Source     : class mixed-scenario sheet, Q4 (no external components)

Scenario:
Four onboard LEDs must toggle independently at different rates:
- PD12 every 200 ms
- PD13 every 400 ms
- PD14 every 600 ms
- PD15 every 1000 ms

Use TIM4 Output Compare interrupt mode. Each channel must schedule its next
compare event by adding its interval to the previous compare value.

Tasks:
1. Configure TIM4 with a 1 ms timer tick.
2. Configure four Output Compare channels in Toggle mode.
3. Enable the TIM4 global interrupt.
4. Start all channels with HAL_TIM_OC_Start_IT().
5. Initialize each CCR to its first interval.
6. In HAL_TIM_OC_DelayElapsedCallback(), toggle the matching onboard LED and
   update its compare value with:

      next_compare = previous_compare + interval_ticks

7. Use __HAL_TIM_SET_COMPARE() or the equivalent HAL function. Allow normal
   counter wrap-around instead of resetting the timer in the callback.
8. Keep each callback short and verify the four rates using debugger values
   and LED observations.

======================================================================
SOLUTION
======================================================================

STEP 1 - CubeMX (.ioc) settings
------------------------------------------------------------
SYS               | Debug = Serial Wire
RCC               | default (HSI 16 MHz)
PD12-PD15         | GPIO_Output
TIM4              | Internal clock, Channel1-4 = Output Compare No Output
TIM4 params       | PSC = 15999 (1 ms), ARR = 65535 (full 16-bit)
OC CH1..CH4       | Mode Frozen/Timing, Pulse 200 / 400 / 600 / 1000
TIM4 NVIC         | TIM4 global interrupt ON
------------------------------------------------------------

STEP 2 - replace the WHOLE Core/Src/main.c with the code below
------------------------------------------------------------

void SystemClock_Config(void)
{
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

    __HAL_RCC_PWR_CLK_ENABLE();
    __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE1);

    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI;
    RCC_OscInitStruct.HSIState = RCC_HSI_ON;
    RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
    RCC_OscInitStruct.PLL.PLLState = RCC_PLL_NONE;
    if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK) Error_Handler();

    RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK
                                | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_HSI;
    RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;
    if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_0) != HAL_OK) Error_Handler();
}

static void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOD_CLK_ENABLE();
    __HAL_RCC_GPIOE_CLK_ENABLE();

    HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12|GPIO_PIN_13|GPIO_PIN_14|GPIO_PIN_15, GPIO_PIN_RESET);
    GPIO_InitStruct.Pin = GPIO_PIN_12|GPIO_PIN_13|GPIO_PIN_14|GPIO_PIN_15;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(GPIOD, &GPIO_InitStruct);

}

static void MX_TIM4_Init(void)
{
    TIM_ClockConfigTypeDef sClockSourceConfig = {0};
    TIM_MasterConfigTypeDef sMasterConfig = {0};
    TIM_OC_InitTypeDef sConfigOC = {0};

    htim4.Instance = TIM4;
    htim4.Init.Prescaler = 15999;
    htim4.Init.CounterMode = TIM_COUNTERMODE_UP;
    htim4.Init.Period = 65535;
    htim4.Init.ClockDivision = TIM_CLOCKDIVISION_DIV1;
    htim4.Init.AutoReloadPreload = TIM_AUTORELOAD_PRELOAD_DISABLE;
    if (HAL_TIM_Base_Init(&htim4) != HAL_OK) Error_Handler();
    sClockSourceConfig.ClockSource = TIM_CLOCKSOURCE_INTERNAL;
    if (HAL_TIM_ConfigClockSource(&htim4, &sClockSourceConfig) != HAL_OK) Error_Handler();
    if (HAL_TIM_OC_Init(&htim4) != HAL_OK) Error_Handler();
    sMasterConfig.MasterOutputTrigger = TIM_TRGO_RESET;
    sMasterConfig.MasterSlaveMode = TIM_MASTERSLAVEMODE_DISABLE;
    if (HAL_TIMEx_MasterConfigSynchronization(&htim4, &sMasterConfig) != HAL_OK) Error_Handler();
    sConfigOC.OCMode = TIM_OCMODE_TIMING;
    sConfigOC.OCPolarity = TIM_OCPOLARITY_HIGH;
    sConfigOC.OCFastMode = TIM_OCFAST_DISABLE;
    sConfigOC.Pulse = 200;
    if (HAL_TIM_OC_ConfigChannel(&htim4, &sConfigOC, TIM_CHANNEL_1) != HAL_OK) Error_Handler();
    sConfigOC.Pulse = 400;
    if (HAL_TIM_OC_ConfigChannel(&htim4, &sConfigOC, TIM_CHANNEL_2) != HAL_OK) Error_Handler();
    sConfigOC.Pulse = 600;
    if (HAL_TIM_OC_ConfigChannel(&htim4, &sConfigOC, TIM_CHANNEL_3) != HAL_OK) Error_Handler();
    sConfigOC.Pulse = 1000;
    if (HAL_TIM_OC_ConfigChannel(&htim4, &sConfigOC, TIM_CHANNEL_4) != HAL_OK) Error_Handler();
}

void HAL_TIM_OC_DelayElapsedCallback(TIM_HandleTypeDef *htim)
{
	if(htim->Instance == TIM4)
	{
		uint32_t ccr =0;
		if(htim->Channel == HAL_TIM_ACTIVE_CHANNEL_1)
		{
			HAL_GPIO_TogglePin(GPIOD, GPIO_PIN_12);
			ccr = __HAL_TIM_GET_COMPARE(htim, TIM_CHANNEL_1);
			__HAL_TIM_SET_COMPARE(htim, TIM_CHANNEL_1, (ccr + 200) & 0xFFFF);
		}
		else if(htim->Channel == HAL_TIM_ACTIVE_CHANNEL_2)
		{
			HAL_GPIO_TogglePin(GPIOD, GPIO_PIN_13);
			ccr = __HAL_TIM_GET_COMPARE(htim, TIM_CHANNEL_2);
			__HAL_TIM_SET_COMPARE(htim, TIM_CHANNEL_2, (ccr + 400) & 0xFFFF);
		}
		else if(htim->Channel == HAL_TIM_ACTIVE_CHANNEL_3)
		{
			HAL_GPIO_TogglePin(GPIOD, GPIO_PIN_14);
			ccr = __HAL_TIM_GET_COMPARE(htim, TIM_CHANNEL_3);
			__HAL_TIM_SET_COMPARE(htim, TIM_CHANNEL_3, (ccr + 600) & 0xFFFF);
		}
		else if(htim->Channel == HAL_TIM_ACTIVE_CHANNEL_4)
		{
			HAL_GPIO_TogglePin(GPIOD, GPIO_PIN_15);
			ccr = __HAL_TIM_GET_COMPARE(htim, TIM_CHANNEL_4);
			__HAL_TIM_SET_COMPARE(htim, TIM_CHANNEL_4, (ccr + 1000) & 0xFFFF);
		}
	}
}
