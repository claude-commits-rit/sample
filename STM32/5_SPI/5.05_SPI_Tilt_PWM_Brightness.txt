5.5  SPI TILT PWM BRIGHTNESS
============================

Board      : STM32F407G-DISC1 (Discovery)
Tool       : STM32CubeIDE, HAL
Clock      : HSI 16 MHz (timer clock 16 MHz) unless stated
Debug      : SYS -> Serial Wire, always
Onboard    : PD12 green, PD13 orange, PD14 red, PD15 blue, PA0 user button
             (HIGH when pressed), LIS3DSH accelerometer on SPI1
             (PA5 SCK, PA6 MISO, PA7 MOSI, PE3 CS), internal ADC temp/VREFINT

Scenario:
The more the board is tilted on the X axis, the brighter the LED on that
side. Positive X -> PD12 (TIM4 CH1), negative X -> PD14 (TIM4 CH3).
Y axis: positive -> PD13 (CH2), negative -> PD15 (CH4).

Tasks:
1. SPI1 reads LIS3DSH X and Y every 50 ms (timer flag), after checking
   WHO_AM_I = 0x3F and writing CTRL_REG4 = 0x67.
2. TIM4 CH1-CH4 PWM at 1 kHz.
3. Map the absolute axis value to CCR 0-999 (with a small dead-band so a flat
   board shows all LEDs off).
4. Watch raw X, Y and the CCRs in Live Expressions.

Doubts to answer:
- Why are two's-complement 16-bit values used for X and Y?
- Why is a dead-band needed?

Simple version used here: read only the HIGH byte of each axis as int8_t
(1 g = about 64). No auto-increment, no 16-bit combining.

======================================================================
SOLUTION
======================================================================

STEP 1 - CubeMX (.ioc) settings
------------------------------------------------------------
SYS               | Debug = Serial Wire
RCC               | default (HSI 16 MHz)
SPI1              | Full-Duplex Master, 8-bit, MSB first (PA5/PA6/PA7)
SPI1 params       | Prescaler 16 (16 MHz/16 = 1 MHz), CPOL High, CPHA 2 Edge
PE3               | GPIO_Output, initial level HIGH (CS)
TIM4              | CH1-CH4 = PWM Generation (PD12-PD15), PSC 15, ARR 999
TIM3              | PSC 15999, ARR 49 (50 ms), NVIC TIM3 global interrupt ON
------------------------------------------------------------

STEP 2 - replace the WHOLE Core/Src/main.c with the code below
------------------------------------------------------------
/* ============================================================
   5.5 - complete main.c for STM32F407G-DISC1
   1. New STM32 project -> Board Selector -> STM32F4DISCOVERY.
      "Initialize all peripherals with their default mode?" -> NO.
   2. In the .ioc apply the CubeMX table of this question, save (Ctrl+S)
      so CubeIDE generates the HAL drivers, msp and interrupt files.
   3. Open Core/Src/main.c, select all, paste this whole file. Build.
   ------------------------------------------------------------

   ============================================================ */
#include "main.h"

/* USER CODE END Includes */

TIM_HandleTypeDef htim4;
TIM_HandleTypeDef htim3;

/* USER CODE END PTD */

/* USER CODE BEGIN PV */
uint8_t tx[2];
volatile uint8_t sample =0;
volatile int8_t x =0;
volatile int8_t y =0;
int ax =0;
int ay =0;
int cx =0;
int cy =0;
/* USER CODE END PV */

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_SPI1_Init();
    MX_TIM4_Init();
    MX_TIM3_Init();

    /* USER CODE BEGIN 2 */
	HAL_TIM_PWM_Start(&htim4, TIM_CHANNEL_1);
	HAL_TIM_PWM_Start(&htim4, TIM_CHANNEL_2);
	HAL_TIM_PWM_Start(&htim4, TIM_CHANNEL_3);
	HAL_TIM_PWM_Start(&htim4, TIM_CHANNEL_4);
	/* CTRL_REG4 = 0x67: 100 Hz, X Y Z on */
	tx[0] = 0x20;
	tx[1] = 0x67;
	HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 0);
	HAL_SPI_Transmit(&hspi1, tx, 2, 100);
	HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 1);

	HAL_TIM_Base_Start_IT(&htim3);
    /* USER CODE END 2 */

    while (1)
    {
    /* USER CODE BEGIN 3 */
	if(sample)
	{
		sample =0;
		/* read only the high byte of each axis: X 0x29, Y 0x2B, Z 0x2D */
		tx[0] = 0x29 | 0x80;
		HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 0);
		HAL_SPI_Transmit(&hspi1, tx, 1, 100);
		HAL_SPI_Receive(&hspi1, (uint8_t*)&x, 1, 100);
		HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 1);
		tx[0] = 0x2B | 0x80;
		HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 0);
		HAL_SPI_Transmit(&hspi1, tx, 1, 100);
		HAL_SPI_Receive(&hspi1, (uint8_t*)&y, 1, 100);
		HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 1);

		ax =x;
		ay =y;
		if(ax < 0)
		{
			ax = -ax;
		}
		if(ay < 0)
		{
			ay = -ay;
		}

		/* tilt -> CCR 0..999, flat below 4, full at 64 */
		if(ax < 4)
		{
			cx =0;
		}
		else if(ax > 64)
		{
			cx =999;
		}
		else
		{
			cx = (ax - 4)*999/60;
		}

		if(ay < 4)
		{
			cy =0;
		}
		else if(ay > 64)
		{
			cy =999;
		}
		else
		{
			cy = (ay - 4)*999/60;
		}

		__HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_1, 0);
		__HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_2, 0);
		__HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_3, 0);
		__HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_4, 0);

		if(x > 0)
		{
			__HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_1, cx);
		}
		else
		{
			__HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_3, cx);
		}
		if(y > 0)
		{
			__HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_2, cy);
		}
		else
		{
			__HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_4, cy);
		}
	}
    /* USER CODE END 3 */
    }
}

void SystemClock_Config(void)
{
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

    __HAL_RCC_PWR_CLK_ENABLE();
    __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE1);

    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI;
    RCC_OscInitStruct.HSIState = RCC_HSI_ON;
    RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
    RCC_OscInitStruct.PLL.PLLState = RCC_PLL_NONE;
    if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK) Error_Handler();

    RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK
                                | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_HSI;
    RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;
    if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_0) != HAL_OK) Error_Handler();
}

static void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOD_CLK_ENABLE();
    __HAL_RCC_GPIOE_CLK_ENABLE();

    HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, GPIO_PIN_SET);      /* CS idle high */
    GPIO_InitStruct.Pin = GPIO_PIN_3;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(GPIOE, &GPIO_InitStruct);

}

static void MX_SPI1_Init(void)
{
    hspi1.Instance = SPI1;
    hspi1.Init.Mode = SPI_MODE_MASTER;
    hspi1.Init.Direction = SPI_DIRECTION_2LINES;
    hspi1.Init.DataSize = SPI_DATASIZE_8BIT;
    hspi1.Init.CLKPolarity = SPI_POLARITY_HIGH;
    hspi1.Init.CLKPhase = SPI_PHASE_2EDGE;
    hspi1.Init.NSS = SPI_NSS_SOFT;
    hspi1.Init.BaudRatePrescaler = SPI_BAUDRATEPRESCALER_16;   /* 16 MHz / 16 = 1 MHz */
    hspi1.Init.FirstBit = SPI_FIRSTBIT_MSB;
    hspi1.Init.TIMode = SPI_TIMODE_DISABLE;
    hspi1.Init.CRCCalculation = SPI_CRCCALCULATION_DISABLE;
    hspi1.Init.CRCPolynomial = 10;
    if (HAL_SPI_Init(&hspi1) != HAL_OK) Error_Handler();
}

static void MX_TIM4_Init(void)
{
    TIM_ClockConfigTypeDef sClockSourceConfig = {0};
    TIM_MasterConfigTypeDef sMasterConfig = {0};
    TIM_OC_InitTypeDef sConfigOC = {0};

    htim4.Instance = TIM4;
    htim4.Init.Prescaler = 15;
    htim4.Init.CounterMode = TIM_COUNTERMODE_UP;
    htim4.Init.Period = 999;
    htim4.Init.ClockDivision = TIM_CLOCKDIVISION_DIV1;
    htim4.Init.AutoReloadPreload = TIM_AUTORELOAD_PRELOAD_DISABLE;
    if (HAL_TIM_Base_Init(&htim4) != HAL_OK) Error_Handler();
    sClockSourceConfig.ClockSource = TIM_CLOCKSOURCE_INTERNAL;
    if (HAL_TIM_ConfigClockSource(&htim4, &sClockSourceConfig) != HAL_OK) Error_Handler();
    if (HAL_TIM_PWM_Init(&htim4) != HAL_OK) Error_Handler();
    sMasterConfig.MasterOutputTrigger = TIM_TRGO_RESET;
    sMasterConfig.MasterSlaveMode = TIM_MASTERSLAVEMODE_DISABLE;
    if (HAL_TIMEx_MasterConfigSynchronization(&htim4, &sMasterConfig) != HAL_OK) Error_Handler();
    sConfigOC.OCMode = TIM_OCMODE_PWM1;
    sConfigOC.OCPolarity = TIM_OCPOLARITY_HIGH;
    sConfigOC.OCFastMode = TIM_OCFAST_DISABLE;
    sConfigOC.Pulse = 0;
    if (HAL_TIM_PWM_ConfigChannel(&htim4, &sConfigOC, TIM_CHANNEL_1) != HAL_OK) Error_Handler();
    sConfigOC.Pulse = 0;
    if (HAL_TIM_PWM_ConfigChannel(&htim4, &sConfigOC, TIM_CHANNEL_2) != HAL_OK) Error_Handler();
    sConfigOC.Pulse = 0;
    if (HAL_TIM_PWM_ConfigChannel(&htim4, &sConfigOC, TIM_CHANNEL_3) != HAL_OK) Error_Handler();
    sConfigOC.Pulse = 0;
    if (HAL_TIM_PWM_ConfigChannel(&htim4, &sConfigOC, TIM_CHANNEL_4) != HAL_OK) Error_Handler();
    HAL_TIM_MspPostInit(&htim4);
}
