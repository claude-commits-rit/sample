6.7  IWDG FAULT RECOVERY CONTROLLER
===================================

Board      : STM32F407G-DISC1 (Discovery)
Tool       : STM32CubeIDE, HAL
Clock      : HSI 16 MHz (timer clock 16 MHz) unless stated
Debug      : SYS -> Serial Wire, always
Onboard    : PD12 green, PD13 orange, PD14 red, PD15 blue, PA0 user button
             (HIGH when pressed), LIS3DSH accelerometer on SPI1
             (PA5 SCK, PA6 MISO, PA7 MOSI, PE3 CS), internal ADC temp/VREFINT

Source     : class mixed-scenario sheet, Q10 (no external components)

Scenario:
Use the Independent Watchdog to recover the system if the main application
stops operating. The onboard LEDs must show normal operation and the reason
for a watchdog reset.

Tasks:
1. Configure the IWDG using the internal LSI clock. Select a documented
   timeout, such as approximately 2 seconds, using the actual or measured LSI
   frequency.
2. Use a periodic timer interrupt to set a heartbeat flag. The main loop may
   refresh the watchdog only when the heartbeat and all required application
   health checks are valid.
3. Keep PD12 ON during healthy operation and periodically toggle PD13 as a
   live-operation indicator.
4. If PA0 is held for at least 3 seconds, enter a deliberate test-fault state
   that stops watchdog refresh. Do not use an external fault switch.
5. On the next startup, inspect the RCC reset flags. If the previous reset
   was caused by IWDG, keep PD14 ON for 3 seconds, then return to normal.
6. Clear the reset flags after recording the cause and expose the reset-cause
   variable to the SWD debugger.
7. Confirm that a healthy application does not reset and that a deliberately
   stalled application recovers automatically.
8. Keep SWD enabled, but document how debugger halting affects the watchdog
   test and do not treat a paused debugger as proof of normal operation.

======================================================================
SOLUTION
======================================================================

STEP 1 - CubeMX (.ioc) settings
------------------------------------------------------------
SYS               | Debug = Serial Wire
RCC               | default (HSI 16 MHz)
PD12-PD15         | GPIO_Output
PA0               | GPIO_Input
IWDG              | Prescaler 32, Reload 1999 (about 2 s)
TIM3              | PSC 15999, ARR 99 (100 ms), NVIC TIM3 global interrupt ON
------------------------------------------------------------

STEP 2 - replace the WHOLE Core/Src/main.c with the code below
------------------------------------------------------------
/* ============================================================
   6.7 - complete main.c for STM32F407G-DISC1
   1. New STM32 project -> Board Selector -> STM32F4DISCOVERY.
      "Initialize all peripherals with their default mode?" -> NO.
   2. In the .ioc apply the CubeMX table of this question, save (Ctrl+S)
      so CubeIDE generates the HAL drivers, msp and interrupt files.
   3. Open Core/Src/main.c, select all, paste this whole file. Build.
   ------------------------------------------------------------
Debugger note: when the CPU is halted the IWDG keeps counting (unless
DBGMCU freeze is set), so a paused debugger causes a reset - not a fault.
   ============================================================ */
#include "main.h"

/* USER CODE BEGIN Includes */

/* USER CODE END Includes */

IWDG_HandleTypeDef hiwdg;
TIM_HandleTypeDef htim3;

/* USER CODE BEGIN PTD */

/* USER CODE END PTD */

/* USER CODE BEGIN PV */
volatile uint8_t heartbeat =0;
volatile uint8_t reset_cause =0;   /* 1 = IWDG reset */
uint32_t beats =0;
uint32_t held =0;
/* USER CODE END PV */

void SystemClock_Config(void);
static void MX_GPIO_Init(void);
static void MX_IWDG_Init(void);
static void MX_TIM3_Init(void);

/* USER CODE BEGIN 0 */

/* USER CODE END 0 */

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_IWDG_Init();
    MX_TIM3_Init();

    /* USER CODE BEGIN 2 */
	if(__HAL_RCC_GET_FLAG(RCC_FLAG_IWDGRST))
	{
		reset_cause =1;
		HAL_GPIO_WritePin(GPIOD, GPIO_PIN_14, 1);
		for(int i=0 ; i<30 ; i++)   /* 3 s, keep refreshing */
		{
			HAL_Delay(100);
			HAL_IWDG_Refresh(&hiwdg);
		}
		HAL_GPIO_WritePin(GPIOD, GPIO_PIN_14, 0);
	}
	__HAL_RCC_CLEAR_RESET_FLAGS();
	HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12, 1);
	HAL_TIM_Base_Start_IT(&htim3);
    /* USER CODE END 2 */

    while (1)
    {
    /* USER CODE BEGIN 3 */
	if(heartbeat)
	{
		heartbeat =0;
		beats++;
		if(beats % 5 == 0)
		{
			HAL_GPIO_TogglePin(GPIOD, GPIO_PIN_13);
		}

		if(HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_0) == 1)
		{
			held++;
		}
		else
		{
			held =0;
		}

		if(held >= 30)   /* 3 s held -> fake fault */
		{
			HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12, 0);
			while(1)
			{
			}
		}
		HAL_IWDG_Refresh(&hiwdg);
	}
    /* USER CODE END 3 */
    }
}

void SystemClock_Config(void)
{
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

    __HAL_RCC_PWR_CLK_ENABLE();
    __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE1);

    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI|RCC_OSCILLATORTYPE_LSI;
    RCC_OscInitStruct.HSIState = RCC_HSI_ON;
    RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
    RCC_OscInitStruct.LSIState = RCC_LSI_ON;
    RCC_OscInitStruct.PLL.PLLState = RCC_PLL_NONE;
    if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK) Error_Handler();

    RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK
                                | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_HSI;
    RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;
    if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_0) != HAL_OK) Error_Handler();
}

static void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOD_CLK_ENABLE();
    __HAL_RCC_GPIOE_CLK_ENABLE();

    HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12|GPIO_PIN_13|GPIO_PIN_14|GPIO_PIN_15, GPIO_PIN_RESET);
    GPIO_InitStruct.Pin = GPIO_PIN_12|GPIO_PIN_13|GPIO_PIN_14|GPIO_PIN_15;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(GPIOD, &GPIO_InitStruct);

    GPIO_InitStruct.Pin = GPIO_PIN_0;
    GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);
}

static void MX_IWDG_Init(void)
{
    hiwdg.Instance = IWDG;
    hiwdg.Init.Prescaler = IWDG_PRESCALER_32;
    hiwdg.Init.Reload = 1999;
    if (HAL_IWDG_Init(&hiwdg) != HAL_OK) Error_Handler();
}

static void MX_TIM3_Init(void)
{
    TIM_ClockConfigTypeDef sClockSourceConfig = {0};
    TIM_MasterConfigTypeDef sMasterConfig = {0};

    htim3.Instance = TIM3;
    htim3.Init.Prescaler = 15999;
    htim3.Init.CounterMode = TIM_COUNTERMODE_UP;
    htim3.Init.Period = 99;
    htim3.Init.ClockDivision = TIM_CLOCKDIVISION_DIV1;
    htim3.Init.AutoReloadPreload = TIM_AUTORELOAD_PRELOAD_DISABLE;
    if (HAL_TIM_Base_Init(&htim3) != HAL_OK) Error_Handler();
    sClockSourceConfig.ClockSource = TIM_CLOCKSOURCE_INTERNAL;
    if (HAL_TIM_ConfigClockSource(&htim3, &sClockSourceConfig) != HAL_OK) Error_Handler();
    sMasterConfig.MasterOutputTrigger = TIM_TRGO_RESET;
    sMasterConfig.MasterSlaveMode = TIM_MASTERSLAVEMODE_DISABLE;
    if (HAL_TIMEx_MasterConfigSynchronization(&htim3, &sMasterConfig) != HAL_OK) Error_Handler();
}

/* USER CODE BEGIN 4 */
void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
	if(htim->Instance == TIM3)
	{
		heartbeat =1;
	}
}
/* USER CODE END 4 */

void Error_Handler(void)
{
    __disable_irq();
    while (1) { }
}
