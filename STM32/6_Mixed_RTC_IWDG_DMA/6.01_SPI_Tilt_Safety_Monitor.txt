6.1  SPI TILT SAFETY MONITOR
============================

Board      : STM32F407G-DISC1 (Discovery)
Tool       : STM32CubeIDE, HAL
Clock      : HSI 16 MHz (timer clock 16 MHz) unless stated
Debug      : SYS -> Serial Wire, always
Onboard    : PD12 green, PD13 orange, PD14 red, PD15 blue, PA0 user button
             (HIGH when pressed), LIS3DSH accelerometer on SPI1
             (PA5 SCK, PA6 MISO, PA7 MOSI, PE3 CS), internal ADC temp/VREFINT

Source     : class mixed-scenario sheet, Q6 (no external components)

Scenario:
Build a safety monitor that detects whether the Discovery board is tilted.
Use the onboard LIS3DSH or LIS302DL accelerometer, depending on the board
revision. No external sensor or alarm device may be used.

Suggested direction indication:
- Positive X tilt -> PD12
- Negative X tilt -> PD14
- Positive Y tilt -> PD13
- Negative Y tilt -> PD15
- Within the level band -> all four LEDs OFF

Tasks:
1. Configure the SPI peripheral, chip-select GPIO and any onboard
   accelerometer interrupt GPIO required by the board schematic.
2. Read and verify the accelerometer WHO_AM_I register before normal
   operation. If the identity is incorrect, keep PD14 ON and enter an error
   state.
3. Sample X, Y and Z acceleration values every 100 ms using a timer-based
   schedule.
4. Establish a level reference at startup or when PA0 is pressed for at
   least one second.
5. Select and document a tilt threshold after observing level and tilted raw
   readings.
6. Turn on the direction LED for the dominant axis and sign.
7. If the tilt exceeds the threshold continuously for 3 seconds, blink all
   four LEDs as an onboard alarm indication.
8. Clear the alarm only after the board remains within the safe range for
   2 continuous seconds.
9. Test level, slow tilt, brief shaking, sustained tilt and an accelerometer
   identification failure. Inspect raw readings through SWD.

Simple version used here: read only the HIGH byte of each axis as int8_t
(1 g = about 64). No auto-increment, no 16-bit combining.

======================================================================
SOLUTION
======================================================================

STEP 1 - CubeMX (.ioc) settings
------------------------------------------------------------
SYS               | Debug = Serial Wire
RCC               | default (HSI 16 MHz)
PD12-PD15         | GPIO_Output
SPI1              | Full-Duplex Master, 8-bit, MSB first (PA5/PA6/PA7)
SPI1 params       | Prescaler 16 (16 MHz/16 = 1 MHz), CPOL High, CPHA 2 Edge
PE3               | GPIO_Output, initial level HIGH (CS)
PA0               | GPIO_Input
TIM3              | PSC 15999, ARR 99 (100 ms), NVIC TIM3 global interrupt ON
------------------------------------------------------------

STEP 2 - replace the WHOLE Core/Src/main.c with the code below
------------------------------------------------------------

SPI_HandleTypeDef hspi1;
TIM_HandleTypeDef htim3;

/* USER CODE BEGIN PTD */

/* USER CODE END PTD */

/* USER CODE BEGIN PV */
uint8_t tx[2];
uint8_t who =0;
volatile uint8_t sample =0;
int8_t x =0;
int8_t y =0;
int8_t z =0;
int8_t refx =0;
int8_t refy =0;
int over =0;
int safe =0;
int held =0;
uint8_t alarm =0;
/* USER CODE END PV */

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_SPI1_Init();
    MX_TIM3_Init();

    /* USER CODE BEGIN 2 */
	/* WHO_AM_I check */
	tx[0] = 0x0F | 0x80;
	HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 0);
	HAL_SPI_Transmit(&hspi1, tx, 1, 100);
	HAL_SPI_Receive(&hspi1, &who, 1, 100);
	HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 1);
	if(who != 0x3F)
	{
		HAL_GPIO_WritePin(GPIOD, GPIO_PIN_14, 1);
		while(1)
		{
		}
	}

	/* CTRL_REG4 = 0x67: 100 Hz, X Y Z on */
	tx[0] = 0x20;
	tx[1] = 0x67;
	HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 0);
	HAL_SPI_Transmit(&hspi1, tx, 2, 100);
	HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 1);

	HAL_Delay(50);
	/* level reference */
	/* read only the high byte of each axis: X 0x29, Y 0x2B, Z 0x2D */
	tx[0] = 0x29 | 0x80;
	HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 0);
	HAL_SPI_Transmit(&hspi1, tx, 1, 100);
	HAL_SPI_Receive(&hspi1, (uint8_t*)&refx, 1, 100);
	HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 1);
	tx[0] = 0x2B | 0x80;
	HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 0);
	HAL_SPI_Transmit(&hspi1, tx, 1, 100);
	HAL_SPI_Receive(&hspi1, (uint8_t*)&refy, 1, 100);
	HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 1);
	HAL_TIM_Base_Start_IT(&htim3);
    /* USER CODE END 2 */

    while (1)
    {
    /* USER CODE BEGIN 3 */
	if(sample)
	{
		sample =0;
		/* read only the high byte of each axis: X 0x29, Y 0x2B, Z 0x2D */
		tx[0] = 0x29 | 0x80;
		HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 0);
		HAL_SPI_Transmit(&hspi1, tx, 1, 100);
		HAL_SPI_Receive(&hspi1, (uint8_t*)&x, 1, 100);
		HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 1);
		tx[0] = 0x2B | 0x80;
		HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 0);
		HAL_SPI_Transmit(&hspi1, tx, 1, 100);
		HAL_SPI_Receive(&hspi1, (uint8_t*)&y, 1, 100);
		HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 1);
		tx[0] = 0x2D | 0x80;
		HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 0);
		HAL_SPI_Transmit(&hspi1, tx, 1, 100);
		HAL_SPI_Receive(&hspi1, (uint8_t*)&z, 1, 100);
		HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, 1);
		x = x - refx;
		y = y - refy;

		/* PA0 held for 1 s (10 samples) = new level reference */
		if(HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_0) == 1)
		{
			held++;
			if(held == 10)
			{
				refx = refx + x;
				refy = refy + y;
			}
		}
		else
		{
			held =0;
		}

		int ax =x;
		int ay =y;
		if(ax < 0)
		{
			ax = -ax;
		}
		if(ay < 0)
		{
			ay = -ay;
		}

		if(ax > 16 || ay > 16)
		{
			over++;
			safe =0;
		}
		else
		{
			safe++;
			over =0;
		}

		if(over >= 30)          /* 3 s tilted */
		{
			alarm =1;
		}
		if(alarm && safe >= 20) /* 2 s safe */
		{
			alarm =0;
		}

		if(alarm)
		{
			HAL_GPIO_TogglePin(GPIOD, GPIO_PIN_12|GPIO_PIN_13|GPIO_PIN_14|GPIO_PIN_15);
		}
		else
		{
			HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12|GPIO_PIN_13|GPIO_PIN_14|GPIO_PIN_15, 0);
			if(over > 0)
			{
				if(ax >= ay)
				{
					if(x > 0)
					{
						HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12, 1);
					}
					else
					{
						HAL_GPIO_WritePin(GPIOD, GPIO_PIN_14, 1);
					}
				}
				else
				{
					if(y > 0)
					{
						HAL_GPIO_WritePin(GPIOD, GPIO_PIN_13, 1);
					}
					else
					{
						HAL_GPIO_WritePin(GPIOD, GPIO_PIN_15, 1);
					}
				}
			}
		}
	}
    /* USER CODE END 3 */
    }
}

void SystemClock_Config(void)
{
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

    __HAL_RCC_PWR_CLK_ENABLE();
    __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE1);

    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI;
    RCC_OscInitStruct.HSIState = RCC_HSI_ON;
    RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
    RCC_OscInitStruct.PLL.PLLState = RCC_PLL_NONE;
    if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK) Error_Handler();

    RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK
                                | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_HSI;
    RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;
    if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_0) != HAL_OK) Error_Handler();
}

static void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOD_CLK_ENABLE();
    __HAL_RCC_GPIOE_CLK_ENABLE();

    HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12|GPIO_PIN_13|GPIO_PIN_14|GPIO_PIN_15, GPIO_PIN_RESET);
    GPIO_InitStruct.Pin = GPIO_PIN_12|GPIO_PIN_13|GPIO_PIN_14|GPIO_PIN_15;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(GPIOD, &GPIO_InitStruct);

    HAL_GPIO_WritePin(GPIOE, GPIO_PIN_3, GPIO_PIN_SET);      /* CS idle high */
    GPIO_InitStruct.Pin = GPIO_PIN_3;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(GPIOE, &GPIO_InitStruct);

    GPIO_InitStruct.Pin = GPIO_PIN_0;
    GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);
}

static void MX_SPI1_Init(void)
{
    hspi1.Instance = SPI1;
    hspi1.Init.Mode = SPI_MODE_MASTER;
    hspi1.Init.Direction = SPI_DIRECTION_2LINES;
    hspi1.Init.DataSize = SPI_DATASIZE_8BIT;
    hspi1.Init.CLKPolarity = SPI_POLARITY_HIGH;
    hspi1.Init.CLKPhase = SPI_PHASE_2EDGE;
    hspi1.Init.NSS = SPI_NSS_SOFT;
    hspi1.Init.BaudRatePrescaler = SPI_BAUDRATEPRESCALER_16;   /* 16 MHz / 16 = 1 MHz */
    hspi1.Init.FirstBit = SPI_FIRSTBIT_MSB;
    hspi1.Init.TIMode = SPI_TIMODE_DISABLE;
    hspi1.Init.CRCCalculation = SPI_CRCCALCULATION_DISABLE;
    hspi1.Init.CRCPolynomial = 10;
    if (HAL_SPI_Init(&hspi1) != HAL_OK) Error_Handler();
}

static void MX_TIM3_Init(void)
{
    TIM_ClockConfigTypeDef sClockSourceConfig = {0};
    TIM_MasterConfigTypeDef sMasterConfig = {0};

    htim3.Instance = TIM3;
    htim3.Init.Prescaler = 15999;
    htim3.Init.CounterMode = TIM_COUNTERMODE_UP;
    htim3.Init.Period = 99;
    htim3.Init.ClockDivision = TIM_CLOCKDIVISION_DIV1;
    htim3.Init.AutoReloadPreload = TIM_AUTORELOAD_PRELOAD_DISABLE;
    if (HAL_TIM_Base_Init(&htim3) != HAL_OK) Error_Handler();
    sClockSourceConfig.ClockSource = TIM_CLOCKSOURCE_INTERNAL;
    if (HAL_TIM_ConfigClockSource(&htim3, &sClockSourceConfig) != HAL_OK) Error_Handler();
    sMasterConfig.MasterOutputTrigger = TIM_TRGO_RESET;
    sMasterConfig.MasterSlaveMode = TIM_MASTERSLAVEMODE_DISABLE;
    if (HAL_TIMEx_MasterConfigSynchronization(&htim3, &sMasterConfig) != HAL_OK) Error_Handler();
}

void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
	if(htim->Instance == TIM3)
	{
		sample =1;
	}
}
