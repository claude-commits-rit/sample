6.9  INPUT CAPTURE REACTION TIME GAME
=====================================

Board      : STM32F407G-DISC1 (Discovery)
Tool       : STM32CubeIDE, HAL
Clock      : HSI 16 MHz (timer clock 16 MHz) unless stated
Debug      : SYS -> Serial Wire, always
Onboard    : PD12 green, PD13 orange, PD14 red, PD15 blue, PA0 user button
             (HIGH when pressed), LIS3DSH accelerometer on SPI1
             (PA5 SCK, PA6 MISO, PA7 MOSI, PE3 CS), internal ADC temp/VREFINT
UART       : USART2, PA2 = TX, PA3 = RX, 115200 8N1, no flow control.
             The Discovery has no USB-serial, so connect a USB-TTL adapter
             (adapter TX -> PA3, adapter RX -> PA2, GND -> GND, 3.3 V logic).

Scenario:
Reaction-time game. After a random delay (1-3 s) PD12 turns ON. The player
presses the user button as fast as possible. The reaction time in ms is
measured with TIM2 CH1 input capture on PA0 and printed over UART.

Tasks:
1. TIM2: PSC 15999 (1 ms tick), ARR maximum, CH1 input capture rising edge.
2. Save the counter value at the moment PD12 turns on.
3. On capture: reaction = capture - start. Print "Reaction: NNN ms".
4. Pressing before PD12 is ON -> print "Too early!" and turn on PD14.
5. Keep the best time and print it after each round.

Doubts to answer:
- Why is input capture more accurate than reading HAL_GetTick() in an
  EXTI callback?
- How do you make the delay "random" without extra hardware?

======================================================================
SOLUTION
======================================================================

STEP 1 - CubeMX (.ioc) settings
------------------------------------------------------------
SYS               | Debug = Serial Wire
RCC               | default (HSI 16 MHz)
PD12, PD14        | GPIO_Output
USART2            | Asynchronous, 115200, 8N1 (PA2 TX, PA3 RX)
USART2 NVIC       | USART2 global interrupt ON (if using _IT)
PA0               | TIM2_CH1
TIM2              | CH1 Input Capture direct, rising, filter 15
TIM2 params       | PSC 15999 (1 ms), ARR 4294967295
TIM2 NVIC         | TIM2 global interrupt ON
------------------------------------------------------------

STEP 2 - replace the WHOLE Core/Src/main.c with the code below
------------------------------------------------------------
/* ============================================================
   6.9 - complete main.c for STM32F407G-DISC1
   1. New STM32 project -> Board Selector -> STM32F4DISCOVERY.
      "Initialize all peripherals with their default mode?" -> NO.
   2. In the .ioc apply the CubeMX table of this question, save (Ctrl+S)
      so CubeIDE generates the HAL drivers, msp and interrupt files.
   3. Open Core/Src/main.c, select all, paste this whole file. Build.
   ------------------------------------------------------------
Input capture latches the counter in hardware at the edge, so ISR latency
does not add error (HAL_GetTick in an EXTI callback would).
   ============================================================ */
#include "main.h"

/* USER CODE BEGIN Includes */
#include<stdio.h>
#include<string.h>
/* USER CODE END Includes */

UART_HandleTypeDef huart2;
TIM_HandleTypeDef htim2;

/* USER CODE BEGIN PTD */

/* USER CODE END PTD */

/* USER CODE BEGIN PV */
volatile uint32_t cap =0;
volatile uint8_t captured =0;
uint32_t best =0xFFFFFFFF;
uint32_t start =0;
uint32_t react =0;
char msg[48];
/* USER CODE END PV */

void SystemClock_Config(void);
static void MX_GPIO_Init(void);
static void MX_USART2_UART_Init(void);
static void MX_TIM2_Init(void);

/* USER CODE BEGIN 0 */

/* USER CODE END 0 */

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_USART2_UART_Init();
    MX_TIM2_Init();

    /* USER CODE BEGIN 2 */
	HAL_TIM_IC_Start_IT(&htim2, TIM_CHANNEL_1);
    /* USER CODE END 2 */

    while (1)
    {
    /* USER CODE BEGIN 3 */
	/* random wait 1-3 s from the running timer value */
	uint32_t wait = 1000 + (__HAL_TIM_GET_COUNTER(&htim2) * 7919) % 2000;
	HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12|GPIO_PIN_14, 0);
	HAL_UART_Transmit(&huart2, (uint8_t*)"Get ready...\r\n", 14, 100);
	captured =0;
	HAL_Delay(wait);

	if(captured)
	{
		HAL_GPIO_WritePin(GPIOD, GPIO_PIN_14, 1);
		HAL_UART_Transmit(&huart2, (uint8_t*)"Too early!\r\n", 12, 100);
	}
	else
	{
		start = __HAL_TIM_GET_COUNTER(&htim2);
		HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12, 1);
		while(captured == 0)
		{
		}
		react = cap - start;
		if(react < best)
		{
			best =react;
		}
		sprintf(msg, "Reaction: %lu ms  Best: %lu ms\r\n", react, best);
		HAL_UART_Transmit(&huart2, (uint8_t*)msg, strlen(msg), 100);
	}
	HAL_Delay(2000);
    /* USER CODE END 3 */
    }
}

void SystemClock_Config(void)
{
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

    __HAL_RCC_PWR_CLK_ENABLE();
    __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE1);

    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI;
    RCC_OscInitStruct.HSIState = RCC_HSI_ON;
    RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
    RCC_OscInitStruct.PLL.PLLState = RCC_PLL_NONE;
    if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK) Error_Handler();

    RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK
                                | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_HSI;
    RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;
    if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_0) != HAL_OK) Error_Handler();
}

static void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOD_CLK_ENABLE();
    __HAL_RCC_GPIOE_CLK_ENABLE();

    HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12|GPIO_PIN_14, GPIO_PIN_RESET);
    GPIO_InitStruct.Pin = GPIO_PIN_12|GPIO_PIN_14;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(GPIOD, &GPIO_InitStruct);

}

static void MX_USART2_UART_Init(void)
{
    huart2.Instance = USART2;
    huart2.Init.BaudRate = 115200;
    huart2.Init.WordLength = UART_WORDLENGTH_8B;
    huart2.Init.StopBits = UART_STOPBITS_1;
    huart2.Init.Parity = UART_PARITY_NONE;
    huart2.Init.Mode = UART_MODE_TX_RX;
    huart2.Init.HwFlowCtl = UART_HWCONTROL_NONE;
    huart2.Init.OverSampling = UART_OVERSAMPLING_16;
    if (HAL_UART_Init(&huart2) != HAL_OK) Error_Handler();
}

static void MX_TIM2_Init(void)
{
    TIM_ClockConfigTypeDef sClockSourceConfig = {0};
    TIM_MasterConfigTypeDef sMasterConfig = {0};
    TIM_IC_InitTypeDef sConfigIC = {0};

    htim2.Instance = TIM2;
    htim2.Init.Prescaler = 15999;
    htim2.Init.CounterMode = TIM_COUNTERMODE_UP;
    htim2.Init.Period = 4294967295;
    htim2.Init.ClockDivision = TIM_CLOCKDIVISION_DIV1;
    htim2.Init.AutoReloadPreload = TIM_AUTORELOAD_PRELOAD_DISABLE;
    if (HAL_TIM_Base_Init(&htim2) != HAL_OK) Error_Handler();
    sClockSourceConfig.ClockSource = TIM_CLOCKSOURCE_INTERNAL;
    if (HAL_TIM_ConfigClockSource(&htim2, &sClockSourceConfig) != HAL_OK) Error_Handler();
    if (HAL_TIM_IC_Init(&htim2) != HAL_OK) Error_Handler();
    sMasterConfig.MasterOutputTrigger = TIM_TRGO_RESET;
    sMasterConfig.MasterSlaveMode = TIM_MASTERSLAVEMODE_DISABLE;
    if (HAL_TIMEx_MasterConfigSynchronization(&htim2, &sMasterConfig) != HAL_OK) Error_Handler();
    sConfigIC.ICPolarity = TIM_INPUTCHANNELPOLARITY_RISING;
    sConfigIC.ICSelection = TIM_ICSELECTION_DIRECTTI;
    sConfigIC.ICPrescaler = TIM_ICPSC_DIV1;
    sConfigIC.ICFilter = 15;
    if (HAL_TIM_IC_ConfigChannel(&htim2, &sConfigIC, TIM_CHANNEL_1) != HAL_OK) Error_Handler();
}

/* USER CODE BEGIN 4 */
void HAL_TIM_IC_CaptureCallback(TIM_HandleTypeDef *htim)
{
	if(htim->Instance == TIM2 && captured == 0)
	{
		cap = HAL_TIM_ReadCapturedValue(htim, TIM_CHANNEL_1);
		captured =1;
	}
}
/* USER CODE END 4 */

void Error_Handler(void)
{
    __disable_irq();
    while (1) { }
}
