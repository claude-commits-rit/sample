Q18. UART PASSWORD LOCK
=======================

Board      : ARIES v2 (THEJAS32 SoC, VEGA ET1031 RISC-V core, 100 MHz)
SDK        : Lyra SDK, #include <vega/hal.h> and <vega/aries.h>, build with make
Onboard    : RGB LED on GPIOB (RGB_RED 8, RGB_GREEN 6, RGB_BLUE 7), ACTIVE-LOW
             -> use RGB_x_SET (on) / RGB_x_RST (off)
             LED_2 (5), LED_3 (4), BTN0 (3), BTN1 (2) active-low (pressed = 0),
             SWITCH0 (0), SWITCH1 (1) on GPIOB
UART       : huart0 over the board USB cable, 115200 8N1 (minicom)
Timers     : htimer0/1/2, LoadCount = MILLIS(t) with #define MILLIS(t) ((t)*100000)

Scenario:
The board is "locked" (red LED ON). The user types a 4-character password
followed by Enter. Correct -> green ON for 5 s, "Unlocked". Wrong -> red
blinks 3 times, "Wrong password". After 3 wrong tries -> locked for 10 s
(blue ON), input ignored.

Tasks:
1. Collect characters into a buffer until '\r'.
2. Compare with strcmp() (add '\0').
3. Use Timer 0 for the 5 s and 10 s timeouts (no long delays).

Doubts:
- How do you ignore input during the lockout without disabling UART?


======================================================================
SOLUTION
======================================================================

Copy everything below into main.c, in the same folder as the
Makefile from this RISC-V folder, then: make  /  make flash
------------------------------------------------------------
#include <vega/hal.h>
#include <vega/aries.h>
#include <string.h>

#define MILLIS(ms) ((uint32_t)((ms) * 100000UL))

/* state: 0 LOCKED (red), 1 OPEN (green 5 s), 2 LOCKOUT (blue 10 s) */
char rx;
char cmd[16];
uint8_t idx =0;
volatile uint8_t got =0;
volatile uint32_t ms =0;
uint8_t state =0;
uint8_t wrong =0;
uint32_t t_start =0;

void UART_RxCpltCallback(UART_Handle_t *huart)
{
	got =1;
}

void TIMER_ElapsedCallback(TIMER_Handle_t *htimer)
{
	if(htimer == &htimer0)
	{
		ms++;
	}
}

int main(void)
{
	GPIO_Init(GPIOB, RGB_RED, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_GREEN, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_BLUE, GPIO_OUT);
	RGB_RED_RST;
	RGB_GREEN_RST;
	RGB_BLUE_RST;
	RGB_RED_SET;
	UART_Init(&huart0);
	htimer0.LoadCount = MILLIS(1);
	TIMER_Init(&htimer0);
	TIMER_Start_IT(&htimer0);
	UART_Receive_IT(&huart0, &rx, 1);
	UART_Transmit(&huart0, "Enter password:\r\n", 17);

	while(1)
	{
		if(state == 1 && ms - t_start > 5000)
		{
			state =0;
			RGB_RED_RST;
			RGB_GREEN_RST;
			RGB_BLUE_RST;
			RGB_RED_SET;
			UART_Transmit(&huart0, "Locked again\r\n", 14);
		}
		else if(state == 2 && ms - t_start > 10000)
		{
			state =0;
			wrong =0;
			RGB_RED_RST;
			RGB_GREEN_RST;
			RGB_BLUE_RST;
			RGB_RED_SET;
			UART_Transmit(&huart0, "Try again\r\n", 11);
		}

		if(got)
		{
			got =0;
			if(state == 2)
			{
				/* lockout: ignore input */
			}
			else if(rx == '\r')
			{
				cmd[idx] = 0;
				idx=0;
				if(strcmp(cmd, "1234") == 0)
				{
					state =1;
					wrong =0;
					t_start =ms;
					RGB_RED_RST;
					RGB_GREEN_RST;
					RGB_BLUE_RST;
					RGB_GREEN_SET;
					UART_Transmit(&huart0, "Unlocked\r\n", 10);
				}
				else
				{
					wrong++;
					UART_Transmit(&huart0, "Wrong password\r\n", 16);
					for(int i=0 ; i<3 ; i++)
					{
						RGB_RED_RST;
						delayms(150);
						RGB_RED_SET;
						delayms(150);
					}
					if(wrong >= 3)
					{
						state =2;
						t_start =ms;
						RGB_RED_RST;
						RGB_GREEN_RST;
						RGB_BLUE_RST;
						RGB_BLUE_SET;
						UART_Transmit(&huart0, "Locked for 10 s\r\n", 17);
					}
				}
			}
			else if(rx != '\n' && idx < 15)
			{
				cmd[idx++] = rx;
			}
			UART_Receive_IT(&huart0, &rx, 1);
		}
	}
}
