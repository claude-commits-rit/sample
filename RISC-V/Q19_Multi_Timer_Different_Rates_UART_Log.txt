Q19. MULTI TIMER DIFFERENT RATES UART LOG
=========================================

Board      : ARIES v2 (THEJAS32 SoC, VEGA ET1031 RISC-V core, 100 MHz)
SDK        : Lyra SDK, #include <vega/hal.h> and <vega/aries.h>, build with make
Onboard    : RGB LED on GPIOB (RGB_RED 8, RGB_GREEN 6, RGB_BLUE 7), ACTIVE-LOW
             -> use RGB_x_SET (on) / RGB_x_RST (off)
             LED_2 (5), LED_3 (4), BTN0 (3), BTN1 (2) active-low (pressed = 0),
             SWITCH0 (0), SWITCH1 (1) on GPIOB
UART       : huart0 over the board USB cable, 115200 8N1 (minicom)
Timers     : htimer0/1/2, LoadCount = MILLIS(t) with #define MILLIS(t) ((t)*100000)

Scenario:
Three timers, three jobs:
    Timer 0 (200 ms) toggles red,
    Timer 1 (1 s) prints "Uptime: N s",
    Timer 2 (5 s) prints how many times red toggled in the last 5 s.

Tasks:
1. All three in periodic interrupt mode, one callback.
2. Identify the timer with (htimer == &htimerN).
3. All printing done in main through flags.

Doubts:
- Why should UART_Transmit not be called inside the timer callback?


======================================================================
SOLUTION
======================================================================

Copy everything below into main.c, in the same folder as the
Makefile from this RISC-V folder, then: make  /  make flash
------------------------------------------------------------
#include <vega/hal.h>
#include <vega/aries.h>
#include <stdio.h>
#include <string.h>

#define MILLIS(ms) ((uint32_t)((ms) * 100000UL))

volatile int toggles =0;
volatile uint8_t sec_flag =0;
volatile uint8_t five_flag =0;
volatile uint32_t uptime =0;
char msg[48];

void TIMER_ElapsedCallback(TIMER_Handle_t *htimer)
{
	if(htimer == &htimer0)
	{
		GPIO_TogglePin(GPIOB, RGB_RED);
		toggles++;
	}
	else if(htimer == &htimer1)
	{
		uptime++;
		sec_flag =1;
	}
	else if(htimer == &htimer2)
	{
		five_flag =1;
	}
}

int main(void)
{
	GPIO_Init(GPIOB, RGB_RED, GPIO_OUT);
	RGB_RED_RST;
	UART_Init(&huart0);

	htimer0.LoadCount = MILLIS(200);
	htimer1.LoadCount = MILLIS(1000);
	htimer2.LoadCount = MILLIS(5000);
	TIMER_Init(&htimer0);
	TIMER_Init(&htimer1);
	TIMER_Init(&htimer2);
	TIMER_Start_IT(&htimer0);
	TIMER_Start_IT(&htimer1);
	TIMER_Start_IT(&htimer2);

	while(1)
	{
		if(sec_flag)
		{
			sec_flag =0;
			sprintf(msg, "Uptime: %lu s\r\n", (unsigned long)uptime);
			UART_Transmit(&huart0, msg, strlen(msg));
		}
		if(five_flag)
		{
			five_flag =0;
			sprintf(msg, "Red toggled %d times in 5 s\r\n", toggles);
			toggles =0;
			UART_Transmit(&huart0, msg, strlen(msg));
		}
	}
}
