Q20. BUTTON LONG PRESS SHORT PRESS
==================================

Board      : ARIES v2 (THEJAS32 SoC, VEGA ET1031 RISC-V core, 100 MHz)
SDK        : Lyra SDK, #include <vega/hal.h> and <vega/aries.h>, build with make
Onboard    : RGB LED on GPIOB (RGB_RED 8, RGB_GREEN 6, RGB_BLUE 7), ACTIVE-LOW
             -> use RGB_x_SET (on) / RGB_x_RST (off)
             LED_2 (5), LED_3 (4), BTN0 (3), BTN1 (2) active-low (pressed = 0),
             SWITCH0 (0), SWITCH1 (1) on GPIOB
UART       : huart0 over the board USB cable, 115200 8N1 (minicom)
Timers     : htimer0/1/2, LoadCount = MILLIS(t) with #define MILLIS(t) ((t)*100000)

Scenario:
BTN0 short press (< 1 s) -> next colour (R -> G -> B). Long press (>= 1 s)
-> all off. Print "Short" or "Long" for every press.

Tasks:
1. Timer 0 at 10 ms; count how long BTN0 is held.
2. Decide short/long when the button is released.
3. Debounce: ignore holds shorter than 30 ms.

Doubts:
- Why measure on release and not on press?


======================================================================
SOLUTION
======================================================================

Copy everything below into main.c, in the same folder as the
Makefile from this RISC-V folder, then: make  /  make flash
------------------------------------------------------------
#include <vega/hal.h>
#include <vega/aries.h>

#define MILLIS(ms) ((uint32_t)((ms) * 100000UL))

volatile int held =0;
volatile int released =0;
volatile uint8_t done =0;
uint8_t color =0;   /* 0 red, 1 green, 2 blue */

void TIMER_ElapsedCallback(TIMER_Handle_t *htimer)
{
	if(htimer == &htimer0)
	{
		if(GPIO_ReadPin(GPIOB, BTN0) == 0)   /* still pressed */
		{
			held++;
		}
		else if(held > 0)                    /* just released */
		{
			released =held;
			held =0;
			done =1;
		}
	}
}

int main(void)
{
	GPIO_Init(GPIOB, RGB_RED, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_GREEN, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_BLUE, GPIO_OUT);
	RGB_RED_RST;
	RGB_GREEN_RST;
	RGB_BLUE_RST;
	GPIO_Init(GPIOB, BTN0, GPIO_IN);
	UART_Init(&huart0);
	htimer0.LoadCount = MILLIS(10);
	TIMER_Init(&htimer0);
	TIMER_Start_IT(&htimer0);

	while(1)
	{
		if(done)
		{
			done =0;
			int t = released*10;   /* ms */
			if(t < 30)
			{
				/* bounce, ignore */
			}
			else if(t < 1000)
			{
				UART_Transmit(&huart0, "Short\r\n", 7);
				color++;
				if(color > 2)
				{
					color =0;
				}
				RGB_RED_RST;
				RGB_GREEN_RST;
				RGB_BLUE_RST;
				if(color == 0)
				{
					RGB_RED_SET;
				}
				else if(color == 1)
				{
					RGB_GREEN_SET;
				}
				else
				{
					RGB_BLUE_SET;
				}
			}
			else
			{
				UART_Transmit(&huart0, "Long\r\n", 6);
				RGB_RED_RST;
				RGB_GREEN_RST;
				RGB_BLUE_RST;
			}
		}
	}
}
