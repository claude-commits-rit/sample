Q23. EVENT COUNTER GPIO INTERRUPT UART
======================================

Board      : ARIES v2 (THEJAS32 SoC, VEGA ET1031 RISC-V core, 100 MHz)
SDK        : Lyra SDK, #include <vega/hal.h> and <vega/aries.h>, build with make
Onboard    : RGB LED on GPIOB (RGB_RED 8, RGB_GREEN 6, RGB_BLUE 7), ACTIVE-LOW
             -> use RGB_x_SET (on) / RGB_x_RST (off)
             LED_2 (5), LED_3 (4), BTN0 (3), BTN1 (2) active-low (pressed = 0),
             SWITCH0 (0), SWITCH1 (1) on GPIOB
UART       : huart0 over the board USB cable, 115200 8N1 (minicom)
Timers     : htimer0/1/2, LoadCount = MILLIS(t) with #define MILLIS(t) ((t)*100000)

Scenario:
Count rising edges on GPIOA pin 3 (external button or jumper to 3.3 V, with
a pull-down). Every second print "Events in last second: N" and the total.
Red ON if more than 5 events in one second.

Tasks:
1. GPIO_IT_Enable(3); GPIO_EXTICallback only increments a counter.
2. Timer 0 at 1 s copies and clears the counter, sets a print flag.
3. Add debounce (ignore edges within 50 ms using a 1 ms tick).

Doubts:
- Why must the counter be volatile?


======================================================================
SOLUTION
======================================================================

Copy everything below into main.c, in the same folder as the
Makefile from this RISC-V folder, then: make  /  make flash
------------------------------------------------------------
#include <vega/hal.h>
#include <vega/aries.h>
#include <stdio.h>
#include <string.h>

#define MILLIS(ms) ((uint32_t)((ms) * 100000UL))

/* Wiring: 3.3V -> button/jumper -> GPIOA pin 3, 10k pull-down to GND */
volatile uint32_t ms =0;
volatile uint32_t last =0;
volatile uint32_t events =0;
volatile uint32_t snap =0;
volatile uint8_t report =0;
uint32_t total =0;
char msg[64];

void GPIO_EXTICallback(uint16_t pin)
{
	if(pin == 3 && ms-last>=50)
	{
		last =ms;
		events++;
	}
}

void TIMER_ElapsedCallback(TIMER_Handle_t *htimer)
{
	if(htimer == &htimer0)
	{
		ms++;
		if(ms % 1000 == 0)
		{
			snap =events;
			events =0;
			report =1;
		}
	}
}

int main(void)
{
	GPIO_Init(GPIOB, RGB_RED, GPIO_OUT);
	RGB_RED_RST;
	UART_Init(&huart0);
	GPIO_Init(GPIOA, 3, GPIO_IN);
	GPIO_IT_Enable(3);
	htimer0.LoadCount = MILLIS(1);
	TIMER_Init(&htimer0);
	TIMER_Start_IT(&htimer0);

	while(1)
	{
		if(report)
		{
			report =0;
			total = total + snap;
			sprintf(msg, "Events in last second: %lu  Total: %lu\r\n", (unsigned long)snap, (unsigned long)total);
			UART_Transmit(&huart0, msg, strlen(msg));
			if(snap > 5)
			{
				RGB_RED_SET;
			}
			else
			{
				RGB_RED_RST;
			}
		}
	}
}
