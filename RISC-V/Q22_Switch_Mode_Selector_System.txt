Q22. SWITCH MODE SELECTOR SYSTEM
================================

Board      : ARIES v2 (THEJAS32 SoC, VEGA ET1031 RISC-V core, 100 MHz)
SDK        : Lyra SDK, #include <vega/hal.h> and <vega/aries.h>, build with make
Onboard    : RGB LED on GPIOB (RGB_RED 8, RGB_GREEN 6, RGB_BLUE 7), ACTIVE-LOW
             -> use RGB_x_SET (on) / RGB_x_RST (off)
             LED_2 (5), LED_3 (4), BTN0 (3), BTN1 (2) active-low (pressed = 0),
             SWITCH0 (0), SWITCH1 (1) on GPIOB
UART       : huart0 over the board USB cable, 115200 8N1 (minicom)
Timers     : htimer0/1/2, LoadCount = MILLIS(t) with #define MILLIS(t) ((t)*100000)

Scenario:
SWITCH0/SWITCH1 select the mode:
    00 -> idle (all off), 01 -> blink red 500 ms,
    10 -> RGB cycle 300 ms, 11 -> UART echo mode (green ON).
Print the mode name whenever it changes.

Tasks:
1. Poll switches in main; detect changes.
2. Timer 0 interrupt drives blink/cycle depending on mode.
3. In mode 11 echo every received character.

Doubts:
- How do you stop the timer effects cleanly when switching mode?


======================================================================
SOLUTION
======================================================================

Copy everything below into main.c, in the same folder as the
Makefile from this RISC-V folder, then: make  /  make flash
------------------------------------------------------------
#include <vega/hal.h>
#include <vega/aries.h>

#define MILLIS(ms) ((uint32_t)((ms) * 100000UL))

/* Assumption: a switch reads 1 when ON.
   mode: 0 IDLE, 1 BLINK RED, 2 RGB CYCLE, 3 UART ECHO */
volatile uint8_t mode =0;
volatile uint8_t step =0;
char rx;
volatile uint8_t got =0;
int prev =-1;

void UART_RxCpltCallback(UART_Handle_t *huart)
{
	got =1;
}

void TIMER_ElapsedCallback(TIMER_Handle_t *htimer)
{
	if(htimer == &htimer0)
	{
		if(mode == 1)
		{
			GPIO_TogglePin(GPIOB, RGB_RED);
		}
		else if(mode == 2)
		{
			step++;
			if(step > 2)
			{
				step =0;
			}
			RGB_RED_RST;
			RGB_GREEN_RST;
			RGB_BLUE_RST;
			if(step == 0)
			{
				RGB_RED_SET;
			}
			else if(step == 1)
			{
				RGB_GREEN_SET;
			}
			else
			{
				RGB_BLUE_SET;
			}
		}
	}
}

int main(void)
{
	GPIO_Init(GPIOB, RGB_RED, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_GREEN, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_BLUE, GPIO_OUT);
	RGB_RED_RST;
	RGB_GREEN_RST;
	RGB_BLUE_RST;
	GPIO_Init(GPIOB, SWITCH0, GPIO_IN);
	GPIO_Init(GPIOB, SWITCH1, GPIO_IN);
	UART_Init(&huart0);
	UART_Receive_IT(&huart0, &rx, 1);

	while(1)
	{
		int sel = GPIO_ReadPin(GPIOB, SWITCH1)*2 + GPIO_ReadPin(GPIOB, SWITCH0);
		if(sel != prev)
		{
			prev =sel;
			TIMER_Stop(&htimer0);
			mode =sel;
			step =0;
			RGB_RED_RST;
			RGB_GREEN_RST;
			RGB_BLUE_RST;
			if(mode == 0)
			{
				UART_Transmit(&huart0, "Mode: IDLE\r\n", 12);
			}
			else if(mode == 1)
			{
				htimer0.LoadCount = MILLIS(500);
				TIMER_Init(&htimer0);
				TIMER_Start_IT(&htimer0);
				UART_Transmit(&huart0, "Mode: BLINK RED\r\n", 17);
			}
			else if(mode == 2)
			{
				htimer0.LoadCount = MILLIS(300);
				TIMER_Init(&htimer0);
				TIMER_Start_IT(&htimer0);
				UART_Transmit(&huart0, "Mode: RGB CYCLE\r\n", 17);
			}
			else
			{
				RGB_GREEN_SET;
				UART_Transmit(&huart0, "Mode: UART ECHO\r\n", 17);
			}
		}

		if(got)
		{
			got =0;
			if(mode == 3)
			{
				UART_Transmit(&huart0, &rx, 1);
			}
			UART_Receive_IT(&huart0, &rx, 1);
		}
		delayms(10);
	}
}
