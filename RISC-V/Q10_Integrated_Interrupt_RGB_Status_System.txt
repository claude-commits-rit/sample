Q10. INTEGRATED INTERRUPT RGB STATUS SYSTEM
===========================================

Board      : ARIES v2 (THEJAS32 SoC, VEGA ET1031 RISC-V core, 100 MHz)
SDK        : Lyra SDK, #include <vega/hal.h> and <vega/aries.h>, build with make
Onboard    : RGB LED on GPIOB (RGB_RED 8, RGB_GREEN 6, RGB_BLUE 7), ACTIVE-LOW
             -> use RGB_x_SET (on) / RGB_x_RST (off)
             LED_2 (5), LED_3 (4), BTN0 (3), BTN1 (2) active-low (pressed = 0),
             SWITCH0 (0), SWITCH1 (1) on GPIOB
UART       : huart0 over the board USB cable, 115200 8N1 (minicom)
Timers     : htimer0/1/2, LoadCount = MILLIS(t) with #define MILLIS(t) ((t)*100000)

Source     : class RISC-V practice sheet, Question 10

Develop a complete RGB status system combining GPIO, UART, Timer 0, and a
GPIO external interrupt.

Hardware configuration:
- RGB LED outputs: GPIOB pins defined by RGB_RED, RGB_GREEN, and RGB_BLUE.
- Push button input: GPIOA pin 3.
- UART: huart0.
- Periodic timer: Timer 0.

Required behavior:

1. Initialize all peripherals and start with all LEDs off.
2. Receive one UART character at a time using UART_Receive_IT().
3. Support these UART commands:
   - r: select red continuously
   - g: select green continuously
   - b: select blue continuously
   - o: turn all LEDs off
   - a: enter automatic mode
   - h: print help
4. In automatic mode, Timer 0 must change the active color every 500 ms in
   the sequence red -> green -> blue -> repeat.
5. Each GPIOA pin 3 button press must cycle through:
   OFF -> RED -> GREEN -> BLUE -> AUTO -> OFF
6. Print a UART status message whenever the mode changes.
7. Keep UART, GPIO, and timer callbacks short. Use volatile event flags and
   perform the main application work in main().
8. Add software button debounce without using a long delay inside an ISR.
9. Ignore CR and LF characters sent by a terminal after a command.

Submission requirements:
- main.c
- Makefile
- terminal output showing at least three UART commands
- a short explanation of the interrupt flow
- a state diagram or table for the five operating modes

Suggested repository references:
- examples/Blinky/blinky.c
- examples/GPIO/GPIO_Interrupt/gpio_int.c
- examples/UART/UART_Interrupt/uart_int.c
- examples/Timer/TimerInterrupt/timer_int.c
- hal/drivers/hal_gpio.c
- hal/drivers/hal_uart.c
- hal/drivers/hal_timer.c
- hal/include/aries.h

SOLUTION
======================================================================

Copy everything below into main.c, in the same folder as the
Makefile from this RISC-V folder, then: make  /  make flash

#include <vega/hal.h>
#include <vega/aries.h>

#define MILLIS(ms) ((uint32_t)((ms) * 100000UL))

/* mode: 0 OFF, 1 RED, 2 GREEN, 3 BLUE, 4 AUTO
   button: OFF -> RED -> GREEN -> BLUE -> AUTO -> OFF
   Wiring: 3.3V -> button -> GPIOA pin 3, 10k pull-down to GND */

char rx;
volatile uint8_t got =0;
volatile uint8_t btn =0;
volatile uint8_t tick =0;
volatile uint32_t ms =0;
uint32_t last =0;
uint8_t mode =0;
uint8_t color =0;     /* auto: 0 red, 1 green, 2 blue */
uint8_t changed =1;   /* 1 = update LED and print mode */

void UART_RxCpltCallback(UART_Handle_t *huart)
{
	got =1;
}

void GPIO_EXTICallback(uint16_t pin)
{
	if(pin == 3 && ms-last>200)
	{
		last =ms;
		btn =1;
	}
}

void TIMER_ElapsedCallback(TIMER_Handle_t *htimer)
{
	if(htimer == &htimer0)
	{
		tick =1;
	}
	else if(htimer == &htimer1)
	{
		ms++;
	}
}

int main(void)
{
	GPIO_Init(GPIOB, RGB_RED, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_GREEN, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_BLUE, GPIO_OUT);
	UART_Init(&huart0);
	GPIO_Init(GPIOA, 3, GPIO_IN);
	GPIO_IT_Enable(3);

	htimer0.LoadCount = MILLIS(500);   /* auto colour step */
	htimer1.LoadCount = MILLIS(1);     /* 1 ms tick for debounce */
	TIMER_Init(&htimer0);
	TIMER_Init(&htimer1);
	TIMER_Start_IT(&htimer0);
	TIMER_Start_IT(&htimer1);
	UART_Receive_IT(&huart0, &rx, 1);

	while(1)
	{
		if(got)
		{
			got =0;
			if(rx == 'r')
			{
				mode =1;
				changed =1;
			}
			else if(rx == 'g')
			{
				mode =2;
				changed =1;
			}
			else if(rx == 'b')
			{
				mode =3;
				changed =1;
			}
			else if(rx == 'o')
			{
				mode =0;
				changed =1;
			}
			else if(rx == 'a')
			{
				mode =4;
				changed =1;
			}
			else if(rx == 'h')
			{
				UART_Transmit(&huart0, "r g b = colour, o = off, a = auto, h = help\r\n", 45);
			}
			else if(rx != '\r' && rx != '\n')
			{
				UART_Transmit(&huart0, "Unknown command, h for help\r\n", 29);
			}
			UART_Receive_IT(&huart0, &rx, 1);
		}

		if(btn)
		{
			btn =0;
			mode++;
			if(mode > 4)
			{
				mode =0;
			}
			changed =1;
		}

		if(tick)
		{
			tick =0;
			if(mode == 4)
			{
				color++;
				if(color > 2)
				{
					color =0;
				}
				RGB_RED_RST;
				RGB_GREEN_RST;
				RGB_BLUE_RST;
				if(color == 0)
				{
					RGB_RED_SET;
				}
				else if(color == 1)
				{
					RGB_GREEN_SET;
				}
				else
				{
					RGB_BLUE_SET;
				}
			}
		}

		if(changed)
		{
			changed =0;
			color =0;
			RGB_RED_RST;
			RGB_GREEN_RST;
			RGB_BLUE_RST;
			if(mode == 0)
			{
				UART_Transmit(&huart0, "Mode: OFF\r\n", 11);
			}
			else if(mode == 1)
			{
				RGB_RED_SET;
				UART_Transmit(&huart0, "Mode: RED\r\n", 11);
			}
			else if(mode == 2)
			{
				RGB_GREEN_SET;
				UART_Transmit(&huart0, "Mode: GREEN\r\n", 13);
			}
			else if(mode == 3)
			{
				RGB_BLUE_SET;
				UART_Transmit(&huart0, "Mode: BLUE\r\n", 12);
			}
			else
			{
				RGB_RED_SET;
				UART_Transmit(&huart0, "Mode: AUTO\r\n", 12);
			}
		}
	}
}
