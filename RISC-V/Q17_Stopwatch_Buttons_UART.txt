Q17. STOPWATCH BUTTONS UART
===========================

Board      : ARIES v2 (THEJAS32 SoC, VEGA ET1031 RISC-V core, 100 MHz)
SDK        : Lyra SDK, #include <vega/hal.h> and <vega/aries.h>, build with make
Onboard    : RGB LED on GPIOB (RGB_RED 8, RGB_GREEN 6, RGB_BLUE 7), ACTIVE-LOW
             -> use RGB_x_SET (on) / RGB_x_RST (off)
             LED_2 (5), LED_3 (4), BTN0 (3), BTN1 (2) active-low (pressed = 0),
             SWITCH0 (0), SWITCH1 (1) on GPIOB
UART       : huart0 over the board USB cable, 115200 8N1 (minicom)
Timers     : htimer0/1/2, LoadCount = MILLIS(t) with #define MILLIS(t) ((t)*100000)

Scenario:
Stopwatch: BTN0 start/stop, BTN1 reset. Timer 0 interrupt every 10 ms.
Print "SS.cc" every 100 ms while running. Green ON when running, red ON when
stopped.

Tasks:
1. Timer callback increments centiseconds only while running.
2. Buttons polled in main with debounce.
3. Printing done in main using a print flag set every 10 timer ticks.

Doubts:
- Why is the print flag better than printing inside the callback?


======================================================================
SOLUTION
======================================================================

Copy everything below into main.c, in the same folder as the
Makefile from this RISC-V folder, then: make  /  make flash
------------------------------------------------------------
#include <vega/hal.h>
#include <vega/aries.h>
#include <stdio.h>
#include <string.h>

#define MILLIS(ms) ((uint32_t)((ms) * 100000UL))

volatile uint32_t cs =0;   /* centiseconds */
volatile uint8_t running =0;
volatile uint8_t ticks =0;
volatile uint8_t show =0;
int p0 =1;
int p1 =1;
char msg[24];

void TIMER_ElapsedCallback(TIMER_Handle_t *htimer)
{
	if(htimer == &htimer0 && running)
	{
		cs++;
		ticks++;
		if(ticks >= 10)   /* every 100 ms */
		{
			ticks =0;
			show =1;
		}
	}
}

int main(void)
{
	GPIO_Init(GPIOB, RGB_RED, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_GREEN, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_BLUE, GPIO_OUT);
	RGB_BLUE_RST;
	GPIO_Init(GPIOB, BTN0, GPIO_IN);
	GPIO_Init(GPIOB, BTN1, GPIO_IN);
	UART_Init(&huart0);
	htimer0.LoadCount = MILLIS(10);
	TIMER_Init(&htimer0);
	TIMER_Start_IT(&htimer0);

	while(1)
	{
		int b0 = GPIO_ReadPin(GPIOB, BTN0);
		int b1 = GPIO_ReadPin(GPIOB, BTN1);
		if(p0 == 1 && b0 == 0)   /* start/stop */
		{
			if(running == 0)
			{
				running =1;
			}
			else
			{
				running =0;
			}
		}
		if(p1 == 1 && b1 == 0)   /* reset */
		{
			cs =0;
			show =1;
		}
		p0 =b0;
		p1 =b1;

		if(running)
		{
			RGB_GREEN_SET;
			RGB_RED_RST;
		}
		else
		{
			RGB_GREEN_RST;
			RGB_RED_SET;
		}

		if(show)
		{
			show =0;
			sprintf(msg, "%02lu.%02lu\r\n", (unsigned long)(cs/100), (unsigned long)(cs%100));
			UART_Transmit(&huart0, msg, strlen(msg));
		}
		delayms(20);
	}
}
