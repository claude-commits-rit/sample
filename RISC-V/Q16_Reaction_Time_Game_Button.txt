Q16. REACTION TIME GAME BUTTON
==============================

Board      : ARIES v2 (THEJAS32 SoC, VEGA ET1031 RISC-V core, 100 MHz)
SDK        : Lyra SDK, #include <vega/hal.h> and <vega/aries.h>, build with make
Onboard    : RGB LED on GPIOB (RGB_RED 8, RGB_GREEN 6, RGB_BLUE 7), ACTIVE-LOW
             -> use RGB_x_SET (on) / RGB_x_RST (off)
             LED_2 (5), LED_3 (4), BTN0 (3), BTN1 (2) active-low (pressed = 0),
             SWITCH0 (0), SWITCH1 (1) on GPIOB
UART       : huart0 over the board USB cable, 115200 8N1 (minicom)
Timers     : htimer0/1/2, LoadCount = MILLIS(t) with #define MILLIS(t) ((t)*100000)

Scenario:
After a random wait, the green LED turns on; the player presses BTN0 as fast
as possible. Measure the reaction time in ms with a 1 ms Timer 0 interrupt
counter and print it. Pressing early prints "Too early!" and lights red.

Tasks:
1. Timer 0 at 1 ms increments a volatile counter.
2. Random wait 1-3 s (use the counter value at the last press as a seed).
3. Poll BTN0 (active-low).
4. Keep and print the best time.

Doubts:
- What is the resolution of your measurement?


======================================================================
SOLUTION
======================================================================

Copy everything below into main.c, in the same folder as the
Makefile from this RISC-V folder, then: make  /  make flash
------------------------------------------------------------
#include <vega/hal.h>
#include <vega/aries.h>
#include <stdio.h>
#include <string.h>

#define MILLIS(ms) ((uint32_t)((ms) * 100000UL))

volatile uint32_t ms =0;
uint32_t best =0xFFFFFFFF;
uint32_t seed =1234;
char msg[48];

void TIMER_ElapsedCallback(TIMER_Handle_t *htimer)
{
	if(htimer == &htimer0)
	{
		ms++;
	}
}

int main(void)
{
	GPIO_Init(GPIOB, RGB_RED, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_GREEN, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_BLUE, GPIO_OUT);
	RGB_RED_RST;
	RGB_GREEN_RST;
	RGB_BLUE_RST;
	GPIO_Init(GPIOB, BTN0, GPIO_IN);
	UART_Init(&huart0);
	htimer0.LoadCount = MILLIS(1);
	TIMER_Init(&htimer0);
	TIMER_Start_IT(&htimer0);

	while(1)
	{
		uint32_t wait = 1000 + (seed % 2000);
		uint32_t t0 =ms;
		uint8_t early =0;
		RGB_RED_RST;
		RGB_GREEN_RST;
		RGB_BLUE_RST;
		UART_Transmit(&huart0, "Get ready...\r\n", 14);

		while(ms - t0 < wait)
		{
			if(GPIO_ReadPin(GPIOB, BTN0) == 0)
			{
				early =1;
				break;
			}
		}

		if(early)
		{
			RGB_RED_SET;
			UART_Transmit(&huart0, "Too early!\r\n", 12);
		}
		else
		{
			RGB_GREEN_SET;
			uint32_t start =ms;
			while(GPIO_ReadPin(GPIOB, BTN0) != 0)
			{
			}
			uint32_t react = ms - start;
			if(react < best)
			{
				best =react;
			}
			sprintf(msg, "Reaction: %lu ms  Best: %lu ms\r\n", (unsigned long)react, (unsigned long)best);
			UART_Transmit(&huart0, msg, strlen(msg));
		}
		seed =ms;
		while(GPIO_ReadPin(GPIOB, BTN0) == 0)   /* wait release */
		{
		}
		delayms(1500);
	}
}
