Q14. UART MENU RGB COLOUR MIXER
===============================

Board      : ARIES v2 (THEJAS32 SoC, VEGA ET1031 RISC-V core, 100 MHz)
SDK        : Lyra SDK, #include <vega/hal.h> and <vega/aries.h>, build with make
Onboard    : RGB LED on GPIOB (RGB_RED 8, RGB_GREEN 6, RGB_BLUE 7), ACTIVE-LOW
             -> use RGB_x_SET (on) / RGB_x_RST (off)
             LED_2 (5), LED_3 (4), BTN0 (3), BTN1 (2) active-low (pressed = 0),
             SWITCH0 (0), SWITCH1 (1) on GPIOB
UART       : huart0 over the board USB cable, 115200 8N1 (minicom)
Timers     : htimer0/1/2, LoadCount = MILLIS(t) with #define MILLIS(t) ((t)*100000)

Scenario:
UART menu that mixes RGB colours:
    '1' red, '2' green, '3' blue, '4' yellow (R+G), '5' cyan (G+B),
    '6' magenta (R+B), '7' white, '0' off, 'm' print menu.

Tasks:
1. Receive one character at a time with UART_Receive_IT().
2. Callback only sets a flag; main decodes and updates the LED.
3. Ignore '\r' and '\n'. Print "Colour: <name>" after each command.

Doubts:
- Why is the receive re-armed after each command?


======================================================================
SOLUTION
======================================================================

Copy everything below into main.c, in the same folder as the
Makefile from this RISC-V folder, then: make  /  make flash
------------------------------------------------------------
#include <vega/hal.h>
#include <vega/aries.h>

char rx;
volatile uint8_t got =0;

void UART_RxCpltCallback(UART_Handle_t *huart)
{
	got =1;
}

int main(void)
{
	GPIO_Init(GPIOB, RGB_RED, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_GREEN, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_BLUE, GPIO_OUT);
	RGB_RED_RST;
	RGB_GREEN_RST;
	RGB_BLUE_RST;
	UART_Init(&huart0);
	UART_Transmit(&huart0, "1 red 2 green 3 blue 4 yellow 5 cyan 6 magenta 7 white 0 off m menu\r\n", 69);
	UART_Receive_IT(&huart0, &rx, 1);

	while(1)
	{
		if(got)
		{
			got =0;
			if(rx == '\r' || rx == '\n')
			{
			}
			else if(rx == 'm')
			{
				UART_Transmit(&huart0, "1 red 2 green 3 blue 4 yellow 5 cyan 6 magenta 7 white 0 off m menu\r\n", 69);
			}
			else if(rx >= '0' && rx <= '7')
			{
				RGB_RED_RST;
				RGB_GREEN_RST;
				RGB_BLUE_RST;
				if(rx == '1')
				{
					RGB_RED_SET;
					UART_Transmit(&huart0, "Colour: red\r\n", 13);
				}
				else if(rx == '2')
				{
					RGB_GREEN_SET;
					UART_Transmit(&huart0, "Colour: green\r\n", 15);
				}
				else if(rx == '3')
				{
					RGB_BLUE_SET;
					UART_Transmit(&huart0, "Colour: blue\r\n", 14);
				}
				else if(rx == '4')
				{
					RGB_RED_SET;
					RGB_GREEN_SET;
					UART_Transmit(&huart0, "Colour: yellow\r\n", 16);
				}
				else if(rx == '5')
				{
					RGB_GREEN_SET;
					RGB_BLUE_SET;
					UART_Transmit(&huart0, "Colour: cyan\r\n", 14);
				}
				else if(rx == '6')
				{
					RGB_RED_SET;
					RGB_BLUE_SET;
					UART_Transmit(&huart0, "Colour: magenta\r\n", 17);
				}
				else if(rx == '7')
				{
					RGB_RED_SET;
					RGB_GREEN_SET;
					RGB_BLUE_SET;
					UART_Transmit(&huart0, "Colour: white\r\n", 15);
				}
				else
				{
					UART_Transmit(&huart0, "Colour: off\r\n", 13);
				}
			}
			else
			{
				UART_Transmit(&huart0, "Invalid\r\n", 9);
			}
			UART_Receive_IT(&huart0, &rx, 1);
		}
	}
}
