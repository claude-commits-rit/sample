Q28. DICE ROLLER BUTTON UART
============================

Board      : ARIES v2 (THEJAS32 SoC, VEGA ET1031 RISC-V core, 100 MHz)
SDK        : Lyra SDK, #include <vega/hal.h> and <vega/aries.h>, build with make
Onboard    : RGB LED on GPIOB (RGB_RED 8, RGB_GREEN 6, RGB_BLUE 7), ACTIVE-LOW
             -> use RGB_x_SET (on) / RGB_x_RST (off)
             LED_2 (5), LED_3 (4), BTN0 (3), BTN1 (2) active-low (pressed = 0),
             SWITCH0 (0), SWITCH1 (1) on GPIOB
UART       : huart0 over the board USB cable, 115200 8N1 (minicom)
Timers     : htimer0/1/2, LoadCount = MILLIS(t) with #define MILLIS(t) ((t)*100000)

Scenario:
Electronic dice. While BTN0 is held a counter runs fast (Timer 0 at 1 ms,
1..6). On release the value is "rolled": print "Dice: N" and show it on the
LEDs (binary on R/G/B). BTN1 prints statistics of all rolls.

Tasks:
1. Timer callback increments the dice value 1-6 while held.
2. Detect release in main; store counts of each face.
3. Debounce.

Doubts:
- Why is this "random" enough for a game?


======================================================================
SOLUTION
======================================================================

Copy everything below into main.c, in the same folder as the
Makefile from this RISC-V folder, then: make  /  make flash
------------------------------------------------------------
#include <vega/hal.h>
#include <vega/aries.h>
#include <stdio.h>
#include <string.h>

#define MILLIS(ms) ((uint32_t)((ms) * 100000UL))

volatile int value =1;
volatile uint8_t rolling =0;
int counts[7];
int p0 =1;
int p1 =1;
char msg[64];

void TIMER_ElapsedCallback(TIMER_Handle_t *htimer)
{
	if(htimer == &htimer0 && rolling)
	{
		value++;
		if(value > 6)
		{
			value =1;
		}
	}
}

int main(void)
{
	GPIO_Init(GPIOB, RGB_RED, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_GREEN, GPIO_OUT);
	GPIO_Init(GPIOB, RGB_BLUE, GPIO_OUT);
	GPIO_Init(GPIOB, BTN0, GPIO_IN);
	GPIO_Init(GPIOB, BTN1, GPIO_IN);
	UART_Init(&huart0);
	htimer0.LoadCount = MILLIS(1);
	TIMER_Init(&htimer0);
	TIMER_Start_IT(&htimer0);

	while(1)
	{
		int b0 = GPIO_ReadPin(GPIOB, BTN0);
		int b1 = GPIO_ReadPin(GPIOB, BTN1);
		if(b0 == 0)
		{
			rolling =1;
		}
		else
		{
			rolling =0;
		}

		if(p0 == 0 && b0 == 1)   /* released */
		{
			int v =value;
			counts[v]++;
			sprintf(msg, "Dice: %d\r\n", v);
			UART_Transmit(&huart0, msg, strlen(msg));
			if(v & 1)
			{
				RGB_RED_SET;
			}
			else
			{
				RGB_RED_RST;
			}
			if(v & 2)
			{
				RGB_GREEN_SET;
			}
			else
			{
				RGB_GREEN_RST;
			}
			if(v & 4)
			{
				RGB_BLUE_SET;
			}
			else
			{
				RGB_BLUE_RST;
			}
		}
		if(p1 == 1 && b1 == 0)
		{
			sprintf(msg, "1:%d 2:%d 3:%d 4:%d 5:%d 6:%d\r\n", counts[1], counts[2], counts[3], counts[4], counts[5], counts[6]);
			UART_Transmit(&huart0, msg, strlen(msg));
		}
		p0 =b0;
		p1 =b1;
		delayms(20);
	}
}
